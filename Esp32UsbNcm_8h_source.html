93 bool send(const uint8_t* frame, uint16_t len) override {
94 if (!tud_ready() || len == 0 || len > CFG_TUD_NET_MTU) return false; // nobody listening
95 // The NCM driver sends 56 bytes of NTB headers plus the frame rounded up
96 // to 4 bytes as one USB transfer. A transfer that is an exact multiple of
97 // the 64 byte packet size needs a terminating zero-length packet, which
98 // wedges the send path on this chip. Trailing zero bytes on an Ethernet
99 // frame are harmless, so such frames are padded.
100 size_t outLen = len;
101 if ((kNtbHeader + ((len + 3u) & ~3u)) % 64 == 0) outLen = len + 4u;
102 uint8_t* copy = (uint8_t*)malloc(outLen);
103 if (!copy) {
104 stats_.txDrops++;
105 return false;
106 }
107 memcpy(copy, frame, len);
108 memset(copy + len, 0, outLen - len);
109 TxItem item = {copy, (uint16_t)outLen};
110 if (xQueueSend(txQueue_, &item, 0) != pdTRUE) {
111 free(copy);
112 stats_.txDrops++; // congested: TCP will retransmit
113 return false;
114 }
115 scheduleDrain();
116 return true;
117 }

159 void onNcmReceive(const uint8_t* src, uint16_t size) {
160 stats_.rxFrames++;
161 if (receiver_) receiver_->onFrame(src, size);
162 // The NCM driver hands over one datagram per tud_network_recv_renew(), and
163 // the call must happen after the receive callback returned. Count the
164 // accepted datagrams and work them off in one deferred call; renew itself
165 // may deliver further datagrams, which just raise the count again.
166 renewCount_++;
167 if (renewActive_) return;
168 renewActive_ = true;
169 usbd_defer_func(&Esp32UsbNcm::renewCb, this, false);
170 }

217 static uint16_t descriptorCb(uint8_t* dst, uint8_t* itf) {
218 Esp32UsbNcm* s = instance();
219 uint8_t itfnum = *itf;
220 *itf += 2; // control + data interface
221 uint8_t strDesc = tinyusb_add_string_descriptor(s->product_.c_str());
222 uint8_t strMac = tinyusb_add_string_descriptor(s->macStr_);
223 uint8_t epData = tinyusb_get_free_duplex_endpoint();
224 uint8_t epNotif = tinyusb_get_free_in_endpoint();
225 if (!strDesc || !strMac || !epData || !epNotif) {
226 TWD_LOGGER.error("USB: no free endpoints or string descriptors for the network interface");
227 return 0;
228 }
229 s->epIn_ = (uint8_t)(0x80 | epData);
230 const uint16_t bulk = TUD_OPT_HIGH_SPEED ? 512 : 64;
231 uint8_t desc[] = {TUD_CDC_NCM_DESCRIPTOR(itfnum, strDesc, strMac, (uint8_t)(0x80 | epNotif), 64, epData,
232 (uint8_t)(0x80 | epData), bulk, s->maxFrame_, 50, 0)};
233 memcpy(dst, desc, sizeof(desc));
234 return sizeof(desc);
235 }

245 // One frame per USB transfer: the NCM driver would otherwise batch frames
246 // that arrive while a transfer is in flight, and the padding rule in send()
247 // only holds for a transfer carrying a single frame.
248 void drain() {
249 drainPending_ = false;
250 TxItem item;
251 uint32_t now = millis();
252 while (xQueuePeek(txQueue_, &item, 0) == pdTRUE) {
253 if (tud_ready()) {
254 if (usbd_edpt_busy(0, epIn_) || !tud_network_can_xmit(item.len)) {
255 stats_.txBusy++;
256 if (!stalled_) {
257 stalled_ = true;
258 stallSince_ = now;
259 } else if (stallTimeoutMs_ && now - stallSince_ > stallTimeoutMs_) {
260 recover();
261 return;
262 }
263 esp_timer_start_once(drainTimer_, 300); // transfer in flight: retry shortly (us)
264 return;
265 }
266 stalled_ = false;
267 tud_network_xmit(item.data, item.len); // copies via tud_network_xmit_cb
268 stats_.txFrames++;
269 } else {
270 stats_.txDrops++;
271 stalled_ = false;
272 }
273 xQueueReceive(txQueue_, &item, 0);
274 free(item.data);
275 }
276 stalled_ = false;
277 }
278
279 // Safety net: if the send path stops making progress, drop the queued frames
280 // and make the host enumerate the device again.
281 void recover() {
282 stats_.resets++;
283 TWD_LOGGER.warning("USB: no send progress for %u ms, dropping %u queued frames and re-enumerating",
284 (unsigned)stallTimeoutMs_, (unsigned)uxQueueMessagesWaiting(txQueue_));
285 TxItem it;
286 while (xQueueReceive(txQueue_, &it, 0) == pdTRUE) free(it.data);
287 stalled_ = false;
288 tud_disconnect();
289 esp_timer_start_once(reconnectTimer_, 800000);
290 }
