5// The core's NCM driver is used at frame level. Frames are exchanged through

240 void pump(async_when_pending_worker_t* w) {
241 // Host -> uplink: this context may call the WiFi chip.
242 Frame* f;
243 while (queue_try_remove(&rxQ_, &f)) {
244 if (receiver_) receiver_->onFrame(f->data, f->len);
245 release(f);
246 }
247
248 // The USB stack is shared with the rest of the core: take its mutex or try again later.
249 if (!mutex_try_enter(&USB.mutex, nullptr)) {
250 w->work_pending = true;
251 return;
252 }
253 while (queue_try_peek(&txQ_, &f)) {
254 if (!tud_ready()) { // host went away: drop what is queued
255 queue_try_remove(&txQ_, &f);
256 release(f);
257 stats_.txDrops++;
258 continue;
259 }
260 if (!tud_network_can_xmit(f->len)) break; // driver busy, try again below
261 tud_network_xmit(f, 0); // copies via tud_network_xmit_cb
262 queue_try_remove(&txQ_, &f);
263 release(f);
264 stats_.txFrames++;
265 }
266 // The driver hands over one datagram per renew.
267 int limit = 10;
268 do {
269 gotRx_ = false;
270 tud_network_recv_renew();
271 tud_task();
272 } while (gotRx_ && --limit > 0);
273 mutex_exit(&USB.mutex);
274
275 if (!queue_is_empty(&txQ_) || !queue_is_empty(&rxQ_)) w->work_pending = true;
276 }
