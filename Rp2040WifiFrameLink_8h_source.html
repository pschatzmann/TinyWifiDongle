124 void maintain() {
125 struct netif* n = __getCYW43Netif();
126 if (!n) {

128 connected_ = false;

131 if (n->input != &Rp2040WifiFrameLink::inputHook) {

133 TWD_LOGGER.info("WiFi station: frames are bridged");
134 }
135 if (dhcp_supplied_address(n) || netif_dhcp_data(n)) dhcp_stop(n); // the host asks the network, not us
136 bool up = netif_is_link_up(n);
137 if (up && !connected_) TWD_LOGGER.info("WiFi associated, forwarding raw frames");
138 if (!up && connected_) TWD_LOGGER.warning("WiFi disconnected");
139 uint32_t now = millis();
140 if (up) {
141 downSince_ = 0;
142 } else if (!downSince_) {

145 TWD_LOGGER.warning("WiFi: not associated with '%s' for %u s (%s)", ssid_.c_str(),
